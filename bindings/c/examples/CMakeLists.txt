cmake_minimum_required(VERSION 3.21)
project(tui_test_c_examples LANGUAGES C CXX)
find_package(tui-test CONFIG REQUIRED)
enable_testing()

foreach(example basic async cpp)
  if(example STREQUAL "cpp")
    add_executable(${example} cpp.cpp)
  else()
    add_executable(${example} ${example}.c)
  endif()
  target_link_libraries(${example} PRIVATE tui-test::ffi)
  target_compile_features(${example} PRIVATE c_std_11 cxx_std_11)
  if(MSVC)
    target_compile_options(${example} PRIVATE /W4 /WX)
  else()
    target_compile_options(${example} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(WIN32)
    add_custom_command(TARGET ${example} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:tui-test::ffi> $<TARGET_FILE_DIR:${example}>)
  elseif(APPLE)
    set_target_properties(${example} PROPERTIES INSTALL_RPATH "@loader_path/../lib")
  else()
    set_target_properties(${example} PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  install(TARGETS ${example} RUNTIME DESTINATION bin)
  add_test(NAME ${example} COMMAND ${example})
  set_tests_properties(${example} PROPERTIES TIMEOUT 30)
endforeach()

if(WIN32)
  install(FILES $<TARGET_FILE:tui-test::ffi> DESTINATION bin)
else()
  install(FILES $<TARGET_FILE:tui-test::ffi> DESTINATION lib)
endif()
